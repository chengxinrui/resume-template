\textbf{Core Publications (First Author)}
\begin{itemize}
    \item \textbf{Physics-Constrained Unsupervised Neural Operator for Long-Horizon PDE Learning on Generalized Geometries} (\textcolor{red}{\textbf{First Author}}, IJCAI CCF-A, 2026.05, \textcolor{red}{\textbf{Accepted}})
    \begin{itemize}[label=\tiny$\bullet$, leftmargin=1em]
        \item \textbf{Research Bottleneck:} Existing unsupervised neural operators rely on differential residuals for physical constraints. High-order spatial derivatives often cause numerical instability and error accumulation on unstructured 3D geometries during long-horizon autoregressive evolution.
        \item \textbf{Core Innovations:} Proposed \textbf{Local Integral Physics Embedding (LIPE)}, which reformulates PDE constraints into an integral evolution form, boosting training stability and reducing VRAM usage without labels. Developed Geometry-Agnostic Latent Adaptation (GALA) to unify unstructured 3D geometry representations, and constructed Gated Spectral Evolution Operator (GSEO) to preserve high-frequency structures.
        \item \textbf{Experimental Metrics:} Reduced drag coefficient error to \textbf{4.06\%} in 3D semi-supervised geometry tasks; achieved \textbf{3.02\%} 20-step error in 2D Navier--Stokes; dropped 20-step error to \textbf{0.09\%} in 1D high-convection scenarios while maintaining stability during $T=50$ extrapolation.
    \end{itemize}
    
    \item \textbf{Integral-Consistent Basis-Universal Neural Operators for Diverse PDE Classes} (\textcolor{red}{\textbf{First Author}}, NeurIPS 2026 CCF-A, \textcolor{blue}{\textbf{Scores: 5/5/4, Rejected}})
    \begin{itemize}[label=\tiny$\bullet$, leftmargin=1em]
        \item \textbf{Research Bottleneck:} Traditional spectral neural operators bind interaction weights to fixed Fourier frequency structures, making it difficult to share dynamic representations across distinct PDE classes (elliptic/parabolic/hyperbolic) and leading to error drift in long-horizon forecasting.
        \item \textbf{Core Innovations:} Proposed a \textbf{Basis-Universal framework} decoupling basis representation from interaction weights, establishing a frequency-index-free complex latent interaction mechanism for unified dynamics modeling. Integrated geometry-aware latent modulation and integral-consistency constraints to enhance cross-geometry generalization.
        \item \textbf{Experimental Metrics:} Reduced $C_d$ error to \textbf{3.95\%} in 3D semi-supervised geometry tasks; achieved \textbf{1.90\%} 20-step error under forced 2D Navier--Stokes; kept 20-step error at the \textbf{0.09\%} scale in 1D convection-diffusion tasks.
    \end{itemize}
\end{itemize}
